
import java.util.LinkedList; // Импортируем LinkedList, который будем использовать для создания очереди
import java.util.Queue;      // Импортируем интерфейс Queue — очередь

class MyStack { // Создаём класс, который будет реализовывать стек

    private Queue<Integer> queue = new LinkedList<>();
    // Создаём очередь для хранения целых чисел
    // private означает, что очередь доступна только внутри класса
    // Integer — тип элементов, которые мы храним

    public MyStack() {
        // Конструктор класса MyStack
        // Вызывается автоматически при создании объекта MyStack
    }

    public void push(int x) {
        // Метод добавляет число x в стек

        queue.offer(x);
        // Добавляем число x в конец очереди

        int size = queue.size();
        // Запоминаем количество элементов в очереди
        // Например, после добавления 3 элементов size будет равен 3

        for (int i = 0; i < size - 1; i++) {
            // Перемещаем все предыдущие элементы в конец очереди
            // Но не перемещаем новый элемент x

            queue.offer(queue.poll());
            // poll() удаляет и возвращает первый элемент очереди
            // offer() добавляет этот элемент в конец очереди
            // Таким образом, старые элементы перемещаются за новый
        }
    }

    public int pop() {
        // Метод удаляет и возвращает верхний элемент стека

        return queue.poll();
        // Удаляем первый элемент очереди и возвращаем его
        // Благодаря изменению порядка при push() это верхушка стека
    }

    public int top() {
        // Метод возвращает верхний элемент стека без удаления

        return queue.peek();
        // peek() возвращает первый элемент очереди, но не удаляет его
    }

    public boolean empty() {
        // Метод проверяет, пустой ли стек

        return queue.isEmpty();
        // Возвращает true, если очередь пуста
        // Возвращает false, если в очереди есть хотя бы один элемент
    }
}
